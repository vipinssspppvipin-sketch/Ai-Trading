const express = require('express'); const path = require('path'); const cors = require('cors'); const axios = require('axios'); const dotenv = require('dotenv'); const { scoreMarketSignal } = require('./src/ai/signalScorer'); dotenv.config(); const app = express(); const PORT = process.env.PORT || 3000; const BASE = process.env.ANGELONE_API_BASE || 'https://apiconnect.angelone.in'; app.use(cors()); app.use(express.json({ limit: '2mb' })); app.use(express.static(path.join(__dirname, 'public'))); function buildHeaders({ apiKey, jwtToken }) { const headers = { 'Content-Type': 'application/json', 'Accept': 'application/json', 'X-UserType': 'USER', 'X-SourceID': 'WEB', 'X-ClientLocalIP': '127.0.0.1', 'X-ClientPublicIP': '127.0.0.1', 'X-MACAddress': '00:00:00:00:00:00', 'X-PrivateKey': apiKey }; if (jwtToken) { headers.Authorization = 'Bearer ' + jwtToken; } return headers; } function generateSampleCandles(symbol = 'SBIN-EQ', points = 30) { const candles = []; let base = 820; let volumeBase = 150000; for (let i = 0; i < points; i++) { const drift = Math.sin(i / 3) * 4.5; const move = (Math.random() - 0.5) * 9; const open = base; const close = Number((open + drift + move).toFixed(2)); const high = Number((Math.max(open, close) + Math.random() * 7 + 1.5).toFixed(2)); const low = Number((Math.min(open, close) - Math.random() * 7 - 1.5).toFixed(2)); const volume = Math.round(volumeBase + Math.random() * 280000 + i * 12000); candles.push({ time: new Date(Date.now() - (points - i) * 60000).toISOString(), symbol, open, high, low, close, volume }); base = close; volumeBase = volume; } return candles; } function computeSma(candles, period) { return candles.map((_, index) => { if (index < period - 1) return null; const slice = candles.slice(index - period + 1, index + 1); const avg = slice.reduce((sum, candle) => sum + Number(candle.close || 0), 0) / slice.length; return Number(avg.toFixed(2)); }); } function computeEma(candles, period) { const values = []; let ema = null; candles.forEach((candle, index) => { const close = Number(candle.close || 0); if (index === 0) { ema = close; values.push(close); return; } const multiplier = 2 / (period + 1); ema = ((close - ema) * multiplier) + ema; values.push(Number(ema.toFixed(2))); }); return values; } app.get('/api/health', (req, res) => { res.json({ ok: true, service: 'ai-trading', charting: true }); }); app.post('/api/login', async (req, res) => { try { const { apiKey, clientCode, password, totp } = req.body || {}; if (!apiKey || !clientCode || !password || !totp) { return res.status(400).json({ success: false, message: 'apiKey, clientCode, password, and totp are required.' }); } const response = await axios.post( `${BASE}/rest/auth/angelbroking/user/v1/loginByPassword`, { clientcode: clientCode, password, totp }, { headers: buildHeaders({ apiKey }) } ); const data = response.data || {}; if (!data.status || !data.data) { return res.status(401).json({ success: false, message: data.message || 'Login failed.' }); } const profileResponse = await axios.get( `${BASE}/rest/secure/angelbroking/user/v1/getProfile`, { headers: buildHeaders({ apiKey, jwtToken: data.data.jwtToken }) } ); const profile = profileResponse.data || {}; return res.json({ success: true, userName: profile.data?.name || clientCode, jwtToken: data.data.jwtToken, refreshToken: data.data.refreshToken || '', feedToken: data.data.feedToken || '' }); } catch (error) { const message = error.response?.data?.message || error.message || 'Login failed.'; return res.status(500).json({ success: false, message }); } }); app.post('/api/quote', async (req, res) => { try { const { apiKey, jwtToken, exchange, token } = req.body || {}; if (!apiKey || !jwtToken || !exchange || !token) { return res.status(400).json({ success: false, message: 'apiKey, jwtToken, exchange, and token are required.' }); } const payload = { mode: 'FULL', exchangeTokens: { [exchange]: [token] } }; const response = await axios.post( `${BASE}/rest/secure/angelbroking/market/v1/quote`, payload, { headers: buildHeaders({ apiKey, jwtToken }) } ); const data = response.data || {}; if (!data.status || !data.data || !data.data.fetched || !data.data.fetched.length) { return res.status(400).json({ success: false, message: data.message || 'Quote fetch failed.' }); } return res.json({ success: true, quote: data.data.fetched[0] }); } catch (error) { const message = error.response?.data?.message || error.message || 'Quote fetch failed.'; return res.status(500).json({ success: false, message }); } }); app.post('/api/ai/signal', (req, res) => { try { const payload = req.body || {}; const signal = scoreMarketSignal({ symbol: payload.symbol || 'UNKNOWN', ltp: payload.ltp, previousClose: payload.previousClose, volume: payload.volume, averageVolume: payload.averageVolume, trendScore: payload.trendScore, momentumScore: payload.momentumScore }); return res.json({ success: true, signal }); } catch (error) { return res.status(500).json({ success: false, message: error.message || 'AI model scoring failed.' }); } }); app.get('/api/chart-data', (req, res) => { const symbol = req.query.symbol || 'SBIN-EQ'; const candles = generateSampleCandles(symbol, 40); const sma9 = computeSma(candles, 9); const sma21 = computeSma(candles, 21); const ema9 = computeEma(candles, 9); res.json({ success: true, symbol, candles, indicators: { sma9, sma21, ema9 } }); }); app.get('*', (req, res) => { res.sendFile(path.join(__dirname, 'public', 'index.html')); }); app.listen(PORT, () => { console.log(`AI Trading server running on http://localhost:${PORT}`); console.log('Charting and indicator layer active.'); });
